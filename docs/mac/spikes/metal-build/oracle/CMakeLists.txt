cmake_minimum_required(VERSION 3.24)
project(oracle LANGUAGES C CXX)
foreach(_o GGML_METAL GGML_BLAS GGML_ACCELERATE GGML_OPENMP GGML_NATIVE BUILD_SHARED_LIBS GGML_BUILD_TESTS GGML_BUILD_EXAMPLES GGML_BACKEND_DL)
  set(${_o} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA}/ggml ${CMAKE_BINARY_DIR}/ggml-oracle EXCLUDE_FROM_ALL)
add_executable(o o.cpp)
target_link_libraries(o PRIVATE ggml-base ggml-cpu)
target_link_options(o PRIVATE LINKER:-no_warn_duplicate_libraries)
